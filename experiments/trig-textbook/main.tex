% ------------------------------------------------------------
% Comparative sidebar — brief excursions outside mathematics
% (numeral systems, calendars, linguistic structure, writing
% systems) illustrating a structural idea under discussion.
% Kept visually subordinate to the main narrative.
% ------------------------------------------------------------
\newtcolorbox{sidebarbox}{
  breakable, enhanced, colback=black!1, colframe=black!30,
  boxrule=0.3pt, arc=0pt, left=8pt, right=8pt, top=5pt, bottom=5pt,
  fontupper=\small
}
\newenvironment{sidebar}[1]
  {\begin{sidebarbox}\noindent\textbf{\small Comparative Note: #1.}\par\smallskip}
  {\end{sidebarbox}}

% ------------------------------------------------------------
% Historical note — brief etymological or biographical asides
% tracing a term or idea to its origin (e.g. the word "calculus",
% or Needham's coinage of "amplitwist"). Distinct in purpose from
% the comparative sidebar above: a comparative note draws a
% structural analogy to a non-mathematical system, while a
% historical note traces the actual lineage of a term or result
% already in use in the main text. Same visual register, kept
% brief and subordinate to the mathematical narrative.
% ------------------------------------------------------------
\newenvironment{historicalnote}[1]
  {\begin{sidebarbox}\noindent\textbf{\small Historical Note: #1.}\par\smallskip}
  {\end{sidebarbox}}
